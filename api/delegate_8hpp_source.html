17template <typename T>
18inline auto threadedContextLoop(runtime::StopToken token, T **out, std::latch *initDone) -> void {
19 T context {};
20 context.install();
21
22 // Init complete
23 *out = &context;
24 initDone->count_down();
25 out = nullptr;
26 initDone = nullptr;
27
28 // Wait for stop token request
29 context.run(token);
30}

32template <typename T>
33inline auto makeThreadedContextImpl() -> std::shared_ptr<T> {
34 struct State {
35 runtime::StopSource source;
36 std::thread thread;
37 };
38
39 State state {};
40 std::latch latch {1};
41 T *context = nullptr;
42 state.thread = std::thread {&threadedContextLoop<T>, state.source.get_token(), &context, &latch};
43 latch.wait(); // Wait init done
44
45 return std::shared_ptr<T> {context, [state = std::move(state)](T *ptr) mutable {
46 ptr->schedule([&]() { // Do the stop request in the thread, more safer :)
47 state.source.request_stop();
48 });
49 state.thread.join();
50 // No-need to delete the context, it is owned by the thread
51 }};
52}
