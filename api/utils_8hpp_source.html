184// The finally awaiter
185// Cleanup is a function or Task<T> to run after the main task is completed
186template <typename T, typename Cleanup>
187class [[ILIAS_CORO_AWAIT_ELIDABLE]] FinallyAwaiter final : public FinallyAwaiterBase {
188public:
189 FinallyAwaiter(TaskHandle<T> main, Cleanup cleanup) : FinallyAwaiterBase(main), mCleanup(std::move(cleanup)) {
190 mOnTaskCompletion = &FinallyAwaiter::onCompletionHelper;
191 }
192
193 auto await_resume() -> T {
194 mException.rethrowIfAny();
195 return unwrapOption(std::move(mValue));
196 }
197private:
198 template <typename U>
199 auto makeCleanup(Task<U> &task) -> TaskHandle<> {
200 return task._leak();
201 }
202
203 template <std::invocable U>
204 auto makeCleanup(U &&fn) -> TaskHandle<> {
205 return fn()._leak();
206 }
207
208 auto onCompletion() -> TaskHandle<> {
209 do {
210 ILIAS_ASSERT(mContext, "The main task should be alive");
211 auto handle = TaskHandle<T>::cast(mContext->task());
212 if (mContext->isStopped()) {
213 break; // Nothing to do
214 }
215 mException = handle.takeException();
216 if (mException) {
217 break;
218 }
219 mValue = makeOption([&]() { return handle.value(); });
220 break;
221 }
222 while (false);
223
224 // Prepare the cleanup task
225 return makeCleanup(mCleanup);
226 }
227
228 static auto onCompletionHelper(FinallyAwaiterBase &_self) -> TaskHandle<> {
229 return static_cast<FinallyAwaiter &>(_self).onCompletion();
230 }
231
232 ExceptionPtr mException;
233 Cleanup mCleanup;
234 Option<T> mValue;
235};
