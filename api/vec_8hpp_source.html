55class IoVec : public detail::IoVecBase {
56public:
57 using Base = detail::IoVecBase;
58
59 constexpr IoVec() : Base{} {}
60 constexpr IoVec(const IoVec &) = default;
61 constexpr IoVec(Buffer buffer) : IoVec{buffer.data(), buffer.size()} {}
62 constexpr IoVec(const void *buf, size_t len) : Base{} {
63 iov_base = const_cast<void *>(buf);
64 iov_len = static_cast<decltype(iov_len)>(len);
65
66#if defined(_WIN32) && !defined(NDEBUG)
67 // In windows, len just uint32_t, we need to check it
68 if (len > maxSize()) {
69 ILIAS_THROW(std::overflow_error("iov_len is too large"));
70 }
71#endif // _WIN32
72 }
73
74 // Get the data pointer (only for const)
75 constexpr auto data() const noexcept -> const std::byte * {
76 return static_cast<const std::byte *>(iov_base);
77 }
78
79 // Get the size of the buffer
80 constexpr auto size() const noexcept -> size_t {
81 return iov_len;
82 }
83
84 // Check if the buffer is empty
85 constexpr auto empty() const noexcept -> bool {
86 return iov_len == 0;
87 }
88
89 // Compare two IoVec
90 constexpr auto operator==(const IoVec &rhs) const noexcept {
91 if (empty() && rhs.empty()) return true;
92 return iov_len == rhs.iov_len && iov_base == rhs.iov_base;
93 }
94
95 // Allow cast to Buffer
96 constexpr operator Buffer() const noexcept {
97 return {data(), size()};
98 }
99
100 // Get the max size allowed by the system
101 constexpr static auto maxSize() noexcept -> size_t {
102 return std::numeric_limits<decltype(iov_len)>::max();
103 }
104};

160// Convert any BufferSequence to IoVecSequence
161template <BufferSequence T>
162inline auto makeIoSequence(const T &seq) {
163 if constexpr (std::convertible_to<T, IoVecSequence>) {
164 return IoVecSequence {seq};
165 }
166 else if constexpr (std::convertible_to<T, MutableIoVecSequence>) {
167 // Mutable -> Const, OK!
168 auto mutSeq = MutableIoVecSequence(seq);
169 return IoVecSequence(static_cast<const IoVec *>(mutSeq.data()), mutSeq.size());
170 }
171 else {
172 std::vector<IoVec> vec;
173 if constexpr (std::ranges::sized_range<T>) {
174 vec.reserve(std::ranges::size(seq));
175 }
176 for (Buffer buf : seq) {
177 vec.emplace_back(buf);
178 }
179 return vec;
180 }
181}
