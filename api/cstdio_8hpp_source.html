106 auto handle() const -> IoResult<IoHandle<fd_t> > & { // Lazy initialization
107 if (mHandle == Err(IoError::Ok)) {
108 fd_t fd {};
109 switch (Kind) {
110 case StdioKind::In: fd = ILIAS_STDIN_FILENO; break;
111 case StdioKind::Out: fd = ILIAS_STDOUT_FILENO; break;
112 case StdioKind::Err: fd = ILIAS_STDERR_FILENO; break;
113 }
114 if (fd == FileDescriptor::invalid()) { // Invalid file descriptor
115 mHandle = Err(IoError::BadFileDescriptor);
116 }
117 else {
118 mHandle = IoHandle<fd_t>::make(fd, IoDescriptor::Tty);

120 }
