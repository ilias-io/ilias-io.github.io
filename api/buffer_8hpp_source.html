194template <std::integral T>
195constexpr auto byteswap(const T value) noexcept -> T { // std::byteswap is C++23, so we need to implement it by ourselves
196 if constexpr (sizeof(T) == 1) {
197 return value;
198 }
199#if defined(__GNUC__) || defined(__clang__)
200 else if constexpr (sizeof(T) == 2) {
201 return std::bit_cast<T>(__builtin_bswap16(std::bit_cast<uint16_t>(value)));
202 }
203 else if constexpr (sizeof(T) == 4) {
204 return std::bit_cast<T>(__builtin_bswap32(std::bit_cast<uint32_t>(value)));
205 }
206 else if constexpr (sizeof(T) == 8) {
207 return std::bit_cast<T>(__builtin_bswap64(std::bit_cast<uint64_t>(value)));
208 }
209#elif defined(_MSC_VER)
210 else if constexpr (sizeof(T) == 2) {
211 if (!std::is_constant_evaluated())
212 return std::bit_cast<T>(_byteswap_ushort(std::bit_cast<uint16_t>(value)));
213 }
214 else if constexpr (sizeof(T) == 4) {
215 if (!std::is_constant_evaluated())
216 return std::bit_cast<T>(_byteswap_ulong(std::bit_cast<uint32_t>(value)));
217 }
218 else if constexpr (sizeof(T) == 8) {
219 if (!std::is_constant_evaluated())
220 return std::bit_cast<T>(_byteswap_uint64(std::bit_cast<uint64_t>(value)));
221 }
222#endif // defined(_MSC_VER)
223
224 // Fallback
225 auto bytes = std::bit_cast<std::array<std::byte, sizeof(T)> >(value);
226 for (size_t i = 0; i < bytes.size() / 2; ++i) {
227 std::swap(bytes[i], bytes[bytes.size() - i - 1]);
228 }
229 return std::bit_cast<T>(bytes);
230}
