117template <typename Fn, typename ...Args> requires (std::invocable<Fn, Args...>)
118class ThreadCallable final : public ThreadState<typename std::invoke_result_t<Fn, Args...>::value_type> {
119public:
120 ThreadCallable(Fn fn, Args &&...args) : mFn(std::move(fn)), mArgs(std::forward<Args>(args)...) {
121 this->mInvoke = &ThreadCallable::onInvoke;
122 this->mDestroy = &ThreadCallable::onDestroy;
123 }
124private:
125 static auto onInvoke(ThreadBase &_self) -> Task<void> {
126 auto &self = static_cast<ThreadCallable &>(_self);
127 if constexpr (std::is_same_v<decltype(self.mValue), Option<void> >) {
128 co_await std::apply(self.mFn, self.mArgs);
129 self.mValue.emplace(); // We are done here
130 }
131 else {
132 self.mValue = co_await std::apply(self.mFn, self.mArgs);
133 }
134 }
135
136 static auto onDestroy(ThreadBase *self) -> void {
137 delete static_cast<ThreadCallable *>(self);
138 }
139
140 Fn mFn;
141 std::tuple<Args...> mArgs;
142};
