311 static auto make(T fd, IoDescriptor::Type type = IoDescriptor::Unknown) -> IoResult<IoHandle<T> > {
312 auto *ctxt = IoContext::currentThread();
313 if (!ctxt) {
314 return Err(IoError::InvalidArgument);
315 }
316 ILIAS_TRY(auto desc, ctxt->addDescriptor(fd_t(fd), type));
317 return IoHandle<T> {
318 std::move(fd),
319 IoDescriptor::Ptr {desc, IoDescriptor::Deleter {ctxt} }
320 };
321 }
322private:

324 IoContext *ctxt = nullptr;

326 auto operator()(IoDescriptor *desc) const -> void { auto _ = ctxt->removeDescriptor(desc); }
327 };
