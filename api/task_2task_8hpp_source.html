248// Environment for the blocking wait task (borrow the task ownship)
249class TaskRefBlockingContext final : private CoroContext {
250public:
251 TaskRefBlockingContext(TaskHandle<> task, CaptureSource source) : CoroContext(std::nostopstate), mTask(task), mSource(source) {
252 auto *executor = runtime::Executor::currentThread();
253 ILIAS_ASSERT(executor, "The current thread has no executor");
254
255 mTask.setCompletionHandler(TaskRefBlockingContext::onComplete);
256 mTask.setContext(*this);
257 this->setExecutor(*executor);
258 this->tracing().pushFrame("wait", source); // TRACING: trace the blocking point
259 }
260 TaskRefBlockingContext(const TaskRefBlockingContext &) = delete;
261
262 // Enter the eventloop and wait for the task to complete
263 auto enter() noexcept -> void {
264 this->tracing().spawn(mSource); // TRACING: blocking wait is also spawn
265 mTask.resume();
266 if (!mTask.done()) {
267 executor().run(mStopExecutor.get_token());
268 }
269 ILIAS_ASSERT(mTask.done(), "??? INTERNAL BUG");
270 }
271
272 template <typename T>
273 auto value() -> T {
274 return TaskHandle<T>::cast(mTask).value();
275 }
276private:
277 static auto onComplete(CoroContext &_self) noexcept -> void { // Break the event loop
278 auto &self = static_cast<TaskRefBlockingContext &>(_self);
279 self.tracing().complete(); // TRACING: completion
280 self.mStopExecutor.request_stop();
281 }
282
283 TaskHandle<> mTask; // The task we use to wait for (borrow)
284 StopSource mStopExecutor; // The stop source of the executor
285 [[ILIAS_NO_UNIQUE_ADDRESS]]
286 CaptureSource mSource; // The source location of the blocking wait
