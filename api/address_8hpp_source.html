227 static auto fromString(std::string_view value) -> Result<IPAddress4, std::errc> {
228 if (value.size() >= INET_ADDRSTRLEN) {
229 return Err(std::errc::invalid_argument);
230 }
231
232 // Parse on our own, for std::string_view, avoid copy
233 std::array<uint8_t, sizeof(::in_addr)> array{};
234 std::size_t index = 0;
235
236 // Split xxx.xxx.xxx.xxx
237 for (const auto [begin, end] : std::views::split(value, '.')) {
238 if (index >= array.size()) { // Too many elements
239 return Err(std::errc::invalid_argument);
240 }
241 std::string_view elem{begin, end};
242 auto [ptr, ec] = std::from_chars(elem.data(), elem.data() + elem.size(), array[index]);
243 if (ec != std::errc{}) {
244 return Err(ec);
245 }
246 if (ptr != elem.data() + elem.size()) { // Not all characters are consumed, it maybe a string like 12a.34.56.78
247 return Err(std::errc::invalid_argument);
248 }
249
250 index += 1;
251 }
252 if (index == array.size()) {
253 return std::bit_cast<IPAddress4>(array);
254 }
255 return Err(std::errc::invalid_argument);
256 }
