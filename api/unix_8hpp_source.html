55 auto bind(UnixEndpoint endpoint) -> IoResult<void> {
56 std::unique_ptr<Char []> path2;
57 if (!endpoint.isAbstract()) {
58 // Convert the endpint(utf8) to wchar_t
59 auto path = endpoint.path();
60#if defined(_WIN32)
61 auto len = ::MultiByteToWideChar(CP_UTF8, 0, path.data(), static_cast<int>(path.size()), nullptr, 0);
62 path2 = std::make_unique<wchar_t[]>(len + 1);
63 ::MultiByteToWideChar(CP_UTF8, 0, path.data(), static_cast<int>(path.size()), path2.get(), len);
64 path2[len] = L'\0';
65#else
66 path2 = std::make_unique<char[]>(path.size() + 1);
67 ::memcpy(path2.get(), path.data(), path.size());
68 path2[path.size()] = '\0';
69#endif // _WIN32
70 }
71 ILIAS_TRYV(mFd.bind(endpoint));
72 mPath.reset(path2.release()); // Put the pointer to the deleter
73 return {};
74 }
75
76 // Operator
77 auto operator =(UnixSocket &&other) -> UnixSocket & = default;
78 auto operator <=>(const UnixSocket &) const = default;
79 auto operator ->() const -> const Socket * { return &mFd; }

82 explicit operator fd_t() const noexcept { return fd_t(mFd); } // NOLINT(modernize-avoid-c-style-cast)

85 ILIAS_TRY(auto sock, Socket::make(AF_UNIX, type, 0));
