201// Because for(xxx; xxx; co_await(++it)) compile failed in some gcc versions.
202// Workaround with this macro
203#define ILIAS_FOR_AWAIT_GCC(var, generator) \
204 if (auto &&_gen_ = (generator); false) {} \
205 else if (bool _first_ = true; false) {} \
206 else \
207 for (auto _it_ = co_await _gen_.begin(); ; _first_ = false) \
208 if (!_first_ ? (co_await (++_it_), 0) : 0; _it_ == _gen_.end()) { \
209 break; \
210 } \
211 else \
212 if (var = *_it_; false) {} \
213 else
214
215// Common version
216#define ILIAS_FOR_AWAIT_GENERIC(var, generator) \
217 if (auto &&_gen_ = (generator); false) {} \
218 else \
219 for (auto _it_ = co_await _gen_.begin(); _it_ != _gen_.end(); co_await (++_it_)) \
220 if (var = *_it_; false) {} \
221 else
