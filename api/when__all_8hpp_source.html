92template <typename ...Ts>
93class WhenAllAwaiter final : public WhenAllAwaiterBase {
94public:
95 using WhenAllAwaiterBase::WhenAllAwaiterBase; // Inherit the constructor
96
97 using Tuple = std::tuple<typename Option<Ts>::value_type ...>; // Using Option to replace void to std::monostate
98 using RawTuple = std::tuple<Ts...>;
99
100 auto await_resume() -> Tuple {
101 mException.rethrowIfAny();
102 return makeResult(std::make_index_sequence<sizeof...(Ts)>{});
103 }
104private:
105 template <size_t I>
106 auto makeResult() -> std::tuple_element_t<I, Tuple> {
107 using Raw = std::tuple_element_t<I, RawTuple>;
108 return *makeOption([&]() {
109 // We replace void to std::monostate in Tuple, but in here, we should use TaskHandle<void>, not std::monostate
110 auto handle = TaskHandle<Raw>::cast(mTasks[I].task());
111 return handle.value();
112 });
113 }
114
115 template <size_t ...Is>
116 auto makeResult(std::index_sequence<Is...>) -> Tuple {
117 return {makeResult<Is>()...};
118 }
119};
