38// Environment for the spawn task common part
39class TaskSpawnContextBase : public RefCounted<TaskSpawnContextBase>, // NOLINT(misc-multiple-inheritance)
40 public ListNode<TaskSpawnContextBase>,
41 protected TaskContext
42{
43public:
44 ILIAS_API
45 TaskSpawnContextBase(TaskHandle<> task, CaptureSource source);
46 TaskSpawnContextBase(const TaskSpawnContextBase &) = delete;
47
48 // Send the stop request of the spawn task
49 using TaskContext::stop;
50 using TaskContext::isStopped;
51
52 // Get the executor of this ctxt
53 using TaskContext::executor;
54
55 // Impl the virtual delete
56 auto operator delete(TaskSpawnContextBase *self, std::destroying_delete_t) -> void {
57 self->mManager(*self, Ops::Delete);
58 }
59
60 // Blocking enter the executor
61 auto enter() -> void {
62 if (!mCompleted) {
63 StopSource source;
64 mCompletionHandler = [&](auto &) noexcept {
65 source.request_stop();
66 };
67 executor().run(source.get_token());
68 }
69 }
70
71 // Get the id of the spawn task
72 auto id() const noexcept -> uintptr_t {
73 return reinterpret_cast<uintptr_t>(this);
74 }
75
76 // Check the spawn task is completed
77 auto isCompleted() const -> bool { return mCompleted; }
78
79 // Set the handle when the task is completed
80 auto setCompletionHandler(SmallFunction<void (TaskSpawnContextBase &) noexcept> handler) -> void {
81 mCompletionHandler = handler;
82 }
83
84 // Set the handler when the task is completed, more convenient
85 template <auto Method, typename Object>
86 auto setCompletionHandler(Object *obj) -> void {
87 mCompletionHandler = [obj](TaskSpawnContextBase &ctxt) noexcept {
88 (obj->*Method)(ctxt);
89 };
90 }
91protected:
92 // Call on the task completed or stopped
93 auto onComplete() noexcept -> void;
94
95 SmallFunction<void (TaskSpawnContextBase &) noexcept> mCompletionHandler; // The completion handler, call when the task is completed or stopped
96 ExceptionPtr mException; // The exception of the task
97 bool mCompleted = false;
98
99 // Virtual method
100 enum Ops : uint8_t { Delete, SetValue };

104// Environment for the spawn task<T>, store the return value
105template <typename T>
106class TaskSpawnContext final : public TaskSpawnContextBase {
107public:
108 TaskSpawnContext(TaskHandle<T> task, CaptureSource source) : TaskSpawnContextBase(task, source) {
109 mManager = TaskSpawnContext::manager;
110 }
111
112 // Virtual delete
113 using TaskSpawnContextBase::operator delete;
114
115 // Get the value of the task, nullopt if the task is stopped
116 auto value() -> Option<T> {
117 ILIAS_ASSERT(mCompleted, "??? INTERNAL BUG");
118 mException.rethrowIfAny();
119 return std::move(mValue);
120 }
121
122 // Allocate / free the spawn task context.
123 auto operator new(size_t size) -> void * {
124 return runtime::allocate(size);
125 }
126
127 auto operator delete(void *ptr, size_t size) noexcept -> void {
128 return runtime::deallocate(ptr, size);
129 }
130protected:
131 static auto manager(TaskSpawnContextBase &_self, Ops op) noexcept -> void {
132 auto &self = static_cast<TaskSpawnContext &>(_self);
133 switch (op) {
134 case Ops::Delete: {
135 self.~TaskSpawnContext();
136 runtime::deallocate(&self, sizeof(self));
137 break;
138 }
139 case Ops::SetValue: {
140 ILIAS_ASSERT(self.isCompleted());
141 auto handle = TaskHandle<T>::cast(self.mTask);
142 if (self.isStopped()) {
143 return;
144 }
145 self.mException = handle.takeException();
146 if (self.mException) {
147 return;
148 }
149 self.mValue = makeOption([&]() { return handle.value(); });
150 return;
151 }
152 }
153 }
154
155 Option<T> mValue;
156};
