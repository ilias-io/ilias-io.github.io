170inline QIoContext::QIoContext(QObject *parent) : QObject(parent) {
171 setObjectName("IliasQIoContext");
172
173 // Handle the timepoint changed
174 mService.setCallback([this](auto timepoint) {
175 if (mTimerId != 0) {
176 killTimer(std::exchange(mTimerId, 0));
177 }
178 if (timepoint) { // Has timers
179 auto now = std::chrono::steady_clock::now();
180 auto diff = std::chrono::duration_cast<std::chrono::nanoseconds>(*timepoint - now);
181 if (diff.count() < 0) { // If already elapsed, set 1ms
182 diff = std::chrono::milliseconds {1};
183 }
184#if QT_VERSION >= QT_VERSION_CHECK(6, 8, 0) // Since Qt 6.8, we can use nanoseconds
185 mTimerId = startTimer(diff, Qt::PreciseTimer);
186#else // Fallback to milliseconds
187 mTimerId = startTimer(diff.count() / 1000'000, Qt::PreciseTimer);
188#endif
189 }
190 });
191}

217inline auto QIoContext::addDescriptor(fd_t fd, IoDescriptor::Type type) -> IoResult<IoDescriptor*> {
218 auto nfd = std::make_unique<QIoDescriptor>(this);
219
220 // If the type is unknown, we need to check it
221 if (type == IoDescriptor::Unknown) {
222 auto ret = fd_utils::type(fd);
223 if (!ret) {
224 return Err(ret.error());
225 }
226 type = *ret;
227 }
228 // Check the fd type that we support
229 switch (type) {
230 case IoDescriptor::Socket:
231 nfd->pollable = true;
232 break;
233
234#if defined(_WIN32)
235 case IoDescriptor::Pipe:
236 case IoDescriptor::File: // Use overlapped io
237 case IoDescriptor::Tty: // Threadpool
238 break;
239#elif defined(__linux__)
240 case IoDescriptor::Pollable:
241 case IoDescriptor::Pipe:
242 nfd->pollable = true; // Linux pipe can be pollable
243 break;
244 case IoDescriptor::Tty:
245 case IoDescriptor::File: // Threadpool
246 break;
247#endif // defined(_WIN32)
248
249 default:
250 ILIAS_WARN("QIo", "QIoContext::addDescriptor(): the descriptor type {} is not supported", type);
251 return Err(IoError::OperationNotSupported);
252 }
253
254 nfd->type = type;
255 nfd->fd = fd;
256 // Prepare env for pollable (in windows only socket can be pollable)
257 if (nfd->pollable) {
258 nfd->poll.readNotifier = new QSocketNotifier(nfd->sockfd, QSocketNotifier::Read, nfd.get());
259 nfd->poll.writeNotifier = new QSocketNotifier(nfd->sockfd, QSocketNotifier::Write, nfd.get());
260 nfd->poll.exceptNotifier = new QSocketNotifier(nfd->sockfd, QSocketNotifier::Exception, nfd.get());
261 nfd->poll.readNotifier->setEnabled(false);
262 nfd->poll.writeNotifier->setEnabled(false);
263 nfd->poll.exceptNotifier->setEnabled(false);
264
265 // Set nonblock, linux pollable fd can also use this way to set nonblock
266 SocketView sockfd{nfd->sockfd};
267 if (auto ret = sockfd.setBlocking(false); !ret) {
268 return Err(ret.error());
269 }
270 }
271
272#if defined(_WIN32)
273 // Setup option for windows specific socket
274 if (nfd->type == IoDescriptor::Socket) {
275 // Disable UDP NetReset and ConnReset
276 SocketView sockfd{nfd->sockfd};
277 if (auto info = sockfd.getOption<sockopt::ProtocolInfo>(); info && info->value().iSocketType == SOCK_DGRAM) {
278 if (auto res = sockfd.setOption(sockopt::UdpConnReset{false}); !res) {
279 ILIAS_WARN("QIo", "QIoContext::addDescriptor(): failed to disable UDP NetReset, {}", res.error().message());
280 }
281 if (auto res = sockfd.setOption(sockopt::UdpNetReset{false}); !res) {
282 ILIAS_WARN("QIo", "QIoContext::addDescriptor(): failed to disable UDP ConnReset, {}", res.error().message());
283 }
284 }
285 // Get sendmsg & recvmsg
286 DWORD bytes;
287 GUID id = WSAID_WSASENDMSG;
288 if (::WSAIoctl(nfd->sockfd, SIO_GET_EXTENSION_FUNCTION_POINTER, &id, sizeof(id), &nfd->sock.sendmsg, sizeof(nfd->sock.sendmsg), &bytes, nullptr, nullptr) == SOCKET_ERROR) {
289 ILIAS_WARN("QIo", "QIoContext::addDescriptor(): failed to get sendmsg, {}", ::GetLastError());
290 }
291 id = WSAID_WSARECVMSG;
292 if (::WSAIoctl(nfd->sockfd, SIO_GET_EXTENSION_FUNCTION_POINTER, &id, sizeof(id), &nfd->sock.recvmsg, sizeof(nfd->sock.recvmsg), &bytes, nullptr, nullptr) == SOCKET_ERROR) {
293 ILIAS_WARN("QIo", "QIoContext::addDescriptor(): failed to get recvmsg, {}", ::GetLastError());
294 }
295 }
296#endif // defined(_WIN32)
297
298 // Set the debug name
299#if !defined(NDEBUG)
300 nfd->setObjectName(QString("IliasQIoDescriptor_%1").arg(nfd->sockfd));
301#endif // !defined(NDEBUG)
302
303 ++mNumOfDescriptors;
304 return nfd.release();
305}

321inline auto QIoContext::read(IoDescriptor *fd, MutableBuffer buffer, std::optional<size_t> offset) -> IoTask<size_t> {
322 auto nfd = static_cast<QIoDescriptor*>(fd);
323
324#if defined(_WIN32)
325 if (nfd->type == IoDescriptor::Pipe || nfd->type == IoDescriptor::File) {
326 QOverlapped overlapped(nfd->fd);
327 ::DWORD bytesRead = 0;
328 if (offset) {
329 overlapped.setOffset(offset.value());
330 }
331 if (::ReadFile(nfd->fd, buffer.data(), buffer.size(), &bytesRead, &overlapped)) {
332 co_return bytesRead;
333 }
334 if (auto err = ::GetLastError(); err != ERROR_IO_PENDING) {
335 co_return Err(SystemError(err));
336 }
337 co_await overlapped;
338 if (::GetOverlappedResult(nfd, &overlapped, &bytesRead, FALSE)) {
339 co_return bytesRead;
340 }
341 auto err = SystemError::fromErrno();
342 if (err == SystemError::Canceled) {
343 co_await this_coro::stopPoint(); // Try set stopped
344 }
345 co_return Err(err);
346 }
347#endif // defined(_WIN32)
348
349#if defined(__linux__)
350 if (nfd->type == IoDescriptor::Pipe) {
351 while (true) {
352 if (auto ret = ::read(nfd->fd, buffer.data(), buffer.size()); ret >= 0) {
353 co_return ret;
354 }
355
356 // Error Handling
357 if (auto err = errno; err != EAGAIN && err != EWOULDBLOCK) {
358 co_return Err(SystemError(err));
359 }
360 if (auto pollret = co_await poll(fd, PollEvent::In); !pollret) {
361 co_return Err(pollret.error());
362 }
363 }
364 }
365
366 if (nfd->type == IoDescriptor::File) {
367 co_return co_await runtime::threadpool::read(nfd->fd, buffer, offset);
368 }
369#endif // defined(__linux__)
370
371 if (nfd->type == IoDescriptor::Tty) {
372 co_return co_await runtime::threadpool::read(nfd->fd, buffer, offset);
373 }
374 if (nfd->type == IoDescriptor::Socket) {
375 co_return co_await recvfrom(fd, buffer, 0, nullptr);
376 }
377 co_return Err(IoError::OperationNotSupported);
378}

455inline auto QIoContext::connect(IoDescriptor *fd, EndpointView endpoint) -> IoTask<void> {
456 auto nfd = static_cast<QIoDescriptor*>(fd);
457 auto sock = SocketView(nfd->sockfd);
458 while (true) {
459 auto ret = sock.connect(endpoint);
460 if (ret) {
461 co_return {};
462 }
463 // Error Handling
464 if (ret.error() != SystemError::WouldBlock && ret.error() != SystemError::InProgress) {
465 co_return Err(ret.error());
466 }
467 auto pollret = co_await poll(fd, PollEvent::Out);
468 if (!pollret) {
469 co_return Err(pollret.error());
470 }
471 auto err = sock.error().value();
472 if (err.isOk()) {
473 co_return {};
474 }
475 co_return Err(err);
476 }

515inline auto QIoContext::sendmsg(IoDescriptor *fd, const MsgHdr &msg, int flags) -> IoTask<size_t> {
516 auto nfd = static_cast<QIoDescriptor*>(fd);
517 auto sendmsg = [&]() -> IoResult<size_t> {
518#if defined(_WIN32)
519 ::DWORD sent = 0;
520 ::WSAMSG copy{msg}; // The WSASENDMSG the msg argument is not const, so copy it
521 if (nfd->sock.sendmsg(nfd->sockfd, &copy, flags, &sent, nullptr, nullptr) != SOCKET_ERROR) {
522 return sent;
523 }
524#else
525 auto ret = ::sendmsg(nfd->sockfd, &msg, flags);
526 if (ret >= 0) {
527 return ret;
528 }
529#endif // defined(_WIN32)
530 return Err(SystemError::fromErrno());
531 };
532 while (true) {
533 auto ret = sendmsg();
534 if (ret) {
535 co_return *ret;
536 }
537 if (ret.error() != SystemError::WouldBlock) {
538 co_return Err(ret.error());
539 }
540 if (auto pollret = co_await poll(fd, PollEvent::Out); !pollret) {
541 co_return Err(pollret.error());
542 }
543 }

566 co_return *ret;

569 co_return Err(ret.error());

615inline auto QPollAwaiter::onNotifierActivated(QSocketDescriptor, QSocketNotifier::Type type) -> void {
616 auto type2str = [](QSocketNotifier::Type type) {
617 switch (type) {
618 case QSocketNotifier::Read: return "Read";
619 case QSocketNotifier::Write: return "Write";
620 case QSocketNotifier::Exception: return "Exception";
621 default: return "Unknown";
622 }
623 };
624 if (mGot) {
625 return;
626 }
627 mGot = true;
628
629 ILIAS_TRACE("QIo", "fd {} was activated by {}", mFd->sockfd, type2str(type));
630 doDisconnect();
631 if (type == QSocketNotifier::Read) {
632 mResult = PollEvent::In;
633 }
634 else if (type == QSocketNotifier::Write) {
635 mResult = PollEvent::Out;
636 }
637 else {
638 mResult = PollEvent::Hup;
639 }
640 mCaller.schedule();
641}

643inline auto QPollAwaiter::doDisconnect() -> void {
644 if (mReadCon) {
645 QObject::disconnect(mReadCon);
646 mFd->poll.numOfRead--;
647 }
648 if (mWriteCon) {
649 QObject::disconnect(mWriteCon);
650 mFd->poll.numOfWrite--;
651 }
652 if (mExceptCon) {
653 QObject::disconnect(mExceptCon);
654 mFd->poll.numOfExcept--;
655 }
656
657 // Check the num of connections, and disable it if 0
658 if (mFd->poll.numOfRead == 0) {
659 mFd->poll.readNotifier->setEnabled(false);
660 }
661 if (mFd->poll.numOfWrite == 0) {
662 mFd->poll.writeNotifier->setEnabled(false);
663 }
664 if (mFd->poll.numOfExcept == 0) {
665 mFd->poll.exceptNotifier->setEnabled(false);
666 }
667}

669inline auto QPollAwaiter::doConnect() -> void {
670 auto fn = [this](QSocketDescriptor fd, QSocketNotifier::Type type) {
671 return onNotifierActivated(fd, type);
672 };
673 if (mEvent & PollEvent::In) {
674 mReadCon = QObject::connect(mFd->poll.readNotifier, &QSocketNotifier::activated, fn);
675 mFd->poll.readNotifier->setEnabled(true);
676 mFd->poll.numOfRead++;
677 }
678 if (mEvent & PollEvent::Out) {
