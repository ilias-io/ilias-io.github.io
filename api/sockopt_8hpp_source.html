75enum OptionAccess : uint8_t {
76 Read = 1 << 0,
77 Write = 1 << 1,
78 ReadWrite = Read | Write,

89template <int Level, int Optname, typename T, OptionAccess Access = OptionAccess::ReadWrite>
90class OptionT {
91public:
92 constexpr OptionT() = default;
93 constexpr OptionT(T value) : mValue(value) {}
94
95 auto setopt(socket_t sock) const -> IoResult<void> requires ((Access & OptionAccess::Write) == OptionAccess::Write) {
96 auto ret = ::setsockopt(sock, Level, Optname, reinterpret_cast<const char *>(&mValue), sizeof(T));
97 if (ret != 0) {
98 return Err(SystemError::fromErrno());
99 }
100 return {};
101 }
102
103 static auto getopt(socket_t sock) -> IoResult<OptionT> requires ((Access & OptionAccess::Read) == OptionAccess::Read) {
104 ::socklen_t optlen = sizeof(T);
105 T value;
106 auto ret = ::getsockopt(sock, Level, Optname, reinterpret_cast<char *>(&value), &optlen);
107 if (ret != 0) {
108 return Err(SystemError::fromErrno());
109 }
110 return OptionT {value};
111 }
112
117 constexpr auto value() const noexcept { return mValue; }
118
124 constexpr operator T() const noexcept { return mValue; }
125private:
126 T mValue {};
127};

140template <DWORD Opcode, typename T, OptionAccess Access = OptionAccess::ReadWrite>
141class WSAOptionT {
142public:
143 constexpr WSAOptionT() = default;
144 constexpr WSAOptionT(T value) : mValue(value) {}
145
146 auto setopt(socket_t sock) const -> IoResult<void> requires ((Access & OptionAccess::Write) == OptionAccess::Write) {
147 DWORD bytes = 0;
148 auto in = mValue;
149 auto out = mValue;
150 auto ret = ::WSAIoctl(sock, Opcode, &in, sizeof(in), &out, sizeof(out), &bytes, nullptr, nullptr);
151 if (ret != 0) {
152 return Err(SystemError::fromErrno());
153 }
154 return {};
155 }
156

158 DWORD bytes = 0;
159 T out{};

161 if (ret != 0) {

165 }
