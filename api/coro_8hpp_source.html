75// MARK: CoroContext
76// The Runtime environment for coroutines.
77//
78// Cancellation invariants:
79// - stop_requested and stopped are different states. stop() only requests
80// cooperative cancellation through the context's StopSource; it does not
81// complete or destroy the coroutine by itself.
82// - A coroutine enters the stopped state only when an awaiter observes the
83// request while the coroutine is suspended and calls CoroHandle::setStopped().
84// - Once isStopped() becomes true, the coroutine must not be resumed or scheduled
85// again. The stopped handler is the single handoff point to the owner/awaiter.
86// - Stopped is not the same as completed: final_suspend and the completion
87// handler are used for normal/exceptional completion, while mStoppedHandler is
88// used for cooperative cancellation.
89// - Contexts built with std::nostopstate are intentionally not cancellable.
90// - Contexts should be Pinned to memory before any coroutine runs
91ILIAS_EXPORT
92class CoroContext {
93public:
94 CoroContext() = default;
95 CoroContext(CoroContext &&) = default;
96 CoroContext(const CoroContext &) = delete;
97 CoroContext(std::nostopstate_t) : mStopSource(std::nostopstate) {}
98
99 // Request to stop the coroutine
100 ILIAS_API
101 auto stop() noexcept -> bool;
102
103 // Set the coroutine to stopped state
104 // See more in CoroHandle::setStopped()
105 ILIAS_API
106 auto setStopped() noexcept -> void;
107
108 // Check if the coroutine is stopped
109 auto isStopped() const noexcept -> bool {
110 return reinterpret_cast<uintptr_t>(mStoppedHandler) == static_cast<uintptr_t>(-1);
111 }

113 auto executor() const noexcept -> Executor & {

199 // co_await for raw awaitable (like std::suspend_never, Task<T>, etc)
200 // We apply the environment on here
201 template <RawAwaitable T, bool Forward = true>
202 auto await_transform(T &&awaitable, [[maybe_unused]] CaptureSource source = {}) -> decltype(auto) { // NOLINT
203 if constexpr (requires { awaitable.setContext(*mContext, source); }) { // It support setContext & with source
204 awaitable.setContext(*mContext, source);
205 }
206 else if constexpr (requires { awaitable.setContext(*mContext); }) { // It support setContext
207 awaitable.setContext(*mContext);
208 }
209#if ILIAS_CORO_TRACE
210 if constexpr (requires { typename T::SkipTracing; }) {
211 return std::forward<T>(awaitable);
212 }
213 else {
214 static_assert(Forward || std::move_constructible<std::decay_t<T> >, "Awaitable must be move_constructible, it will be moved to the awaiter");
215 return TracingAwaitable<T, Forward> { std::forward<T>(awaitable), mContext->tracing(), source }; // Wrap it with tracing
216 }
217#else
218 return std::forward<T>(awaitable);
219#endif // ILIAS_CORO_TRACE
220 }
221
222 // co_await for can be converted to raw awaitable
223 template <IntoRawAwaitable T>
224 auto await_transform(T &&object, CaptureSource source = {}) { // NOLINT
225 auto awaitable = IntoRawAwaitableTrait<T>::into(std::forward<T>(object));
226 return await_transform<decltype(awaitable), false>(std::move(awaitable), source); // Move into inner if necessary
227 }

262 if (mCompletionHandler) {
263 mCompletionHandler(*mContext);
264 }
265#if ILIAS_CORO_TRACE
266 // TRACING: Cleanup the frame belong to us
267 mContext->tracing().popFrame();
268#endif // ILIAS_CORO_TRACE
269 return mPrevAwaiting;
270 }

276protected: // protected ...
277 [[ILIAS_NO_UNIQUE_ADDRESS]] // The CaptureSource will be std::monostate if disabled, so add it
278 CaptureSource mCreation; // The source of the coroutine creation
279 std::coroutine_handle<> mPrevAwaiting = std::noop_coroutine(); // write by Generator :(
