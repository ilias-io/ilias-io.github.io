165 auto setBlocking(bool blocking) const -> IoResult<void> {
166
167#if defined(_WIN32)
168 ::u_long block = blocking ? 0 : 1;
169 ::DWORD bytes = 0;
170 if (::WSAIoctl(mFd, FIONBIO, &block, sizeof(block), nullptr, 0, &bytes, nullptr, nullptr) != 0) {
171 return Err(SystemError::fromErrno());
172 }
173 return {};
174#else
175 int flags = ::fcntl(mFd, F_GETFL, 0);
176 if (flags < 0) {
177 return Err(SystemError::fromErrno());
178 }
179 if (blocking) {
180 flags &= ~O_NONBLOCK;
181 }
182 else {
183 flags |= O_NONBLOCK;
184 }
185 if (::fcntl(mFd, F_SETFL, flags) < 0) {
186 return Err(SystemError::fromErrno());
187 }
188 return {};
189#endif // _WIN32
