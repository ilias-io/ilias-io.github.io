86// The callable of the fiber, for store the fn, args and the return value, it should be allocated on heap
87template <typename Fn, typename ...Args>
88class FiberCallable final : public FiberEntry {
89public:
90 using T = std::invoke_result_t<Fn, Args...>;
91
92 FiberCallable(Fn fn, Args ...args) : mFn(std::forward<Fn>(fn)), mArgs(std::forward<Args>(args)...) {
93 // Setup vtable
94 this->invoke = &FiberCallable::onInvoke;
95 this->destroy = &FiberCallable::onDestroy;
96 }
97 FiberCallable(const FiberCallable &) = delete;
98private:
99 static auto onInvoke(FiberEntry *_self) -> void * {
100 auto self = static_cast<FiberCallable *>(_self);
101 self->mValue = makeOption([&]() {
102 return std::apply(self->mFn, self->mArgs);
103 });
104 return &self->mValue;
105 }
106
107 static auto onDestroy(FiberEntry *_self) -> void {
108 delete static_cast<FiberCallable *>(_self);
109 }
110
111 // State
112 Option<T> mValue; // Put the result here
113 [[ILIAS_NO_UNIQUE_ADDRESS]]
114 Fn mFn;
115 [[ILIAS_NO_UNIQUE_ADDRESS]] // The args may empty
116 std::tuple<Args...> mArgs;
117};

223template <typename T>
224class Fiber {
225public:
226 Fiber() = default;
227 Fiber(Fiber &&) = default;
228
237 template <typename Fn, typename ...Args>
238 requires (std::invocable<Fn, Args...>)
239 explicit Fiber(Fn fn, Args ...args) {
240 auto callable = new fiber::FiberCallable<Fn, Args...> {std::forward<Fn>(fn), std::forward<Args>(args)...};
241 mHandle.reset(fiber::FiberContext::create4(callable));
242 }
243
249 auto wait(runtime::CaptureSource where = {}) -> T {
250 auto executor = runtime::Executor::currentThread();
251 ILIAS_ASSERT(mHandle, "Can't wait for an invalid fiber");
252 ILIAS_ASSERT(executor, "Can't wait for a fiber without an executor");
253
254 auto handle = std::exchange(mHandle, nullptr);
255 handle->setExecutor(*executor);
256 handle->wait(where);
257 return handle->value<T>();
258 }
259
260 // Swap the fiber with another fiber
261 auto swap(Fiber &other) noexcept -> void {
262 mHandle.swap(other.mHandle);
263 }
264
265 // Set the context of the fiber, call on await_transform
266 auto setContext(runtime::CoroContext &ctxt) -> void {
267 mHandle->setExecutor(ctxt.executor());
268 }
269
270 // Operator
271 auto operator =(Fiber &&) -> Fiber & = default;
272
273 // co_await
274 auto operator co_await() && -> fiber::FiberAwaiter<T> {
275 ILIAS_ASSERT(mHandle, "Can't co_wait an invalid fiber");
276 return {std::move(mHandle)};
277 }
278
279 // Check the fiber is valid
280 explicit operator bool() const noexcept {
281 return bool(mHandle);
282 }
283private:
284 fiber::FiberHandle mHandle;
285};
