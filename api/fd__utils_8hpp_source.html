47inline auto pipe() -> IoResult<PipePair> {
48
49#if defined(_WIN32)
50 ::HANDLE reader{};
51 ::HANDLE writer{};
52 if (!win32::pipe(&reader, &writer)) {
53 return Err(SystemError::fromErrno());
54 }
55#else
56 int fds[2];
57 if (::pipe2(fds, O_CLOEXEC | O_NONBLOCK) != 0) {
58 return Err(SystemError::fromErrno());
59 }
60 int reader = fds[0];
61 int writer = fds[1];
62#endif // defined(_WIN32)
63
64 return PipePair {
65 .writer = FileDescriptor{writer},
66 .reader = FileDescriptor{reader}
67 };
68}

109inline auto type(fd_t fd) -> IoResult<IoDescriptor::Type> {
110
111#if defined(_WIN32)
112 ::DWORD type = ::GetFileType(fd);
113 switch (type) {
114 default: break;
115 case FILE_TYPE_CHAR: return IoDescriptor::Tty;
116 case FILE_TYPE_DISK: return IoDescriptor::File;
117 case FILE_TYPE_PIPE: {
118 // Check if it's a pipe or a socket
119 ::DWORD flags = 0;
120 if (!::GetNamedPipeInfo(fd, &flags, nullptr, nullptr, nullptr)) {
121 return IoDescriptor::Socket;
122 }
123 return IoDescriptor::Pipe;
124 }
125 }
126#else
127 struct ::stat st;
128 if (::fstat(fd, &st) == 0) {
129 if (S_ISCHR(st.st_mode) && ::isatty(fd)) {
130 return IoDescriptor::Tty;
131 }
132 if (S_ISREG(st.st_mode)) {
133 return IoDescriptor::File;
134 }
135 if (S_ISFIFO(st.st_mode)) {
136 return IoDescriptor::Pipe;
137 }
138 if (S_ISSOCK(st.st_mode)) {
139 return IoDescriptor::Socket;
140 }
141 return IoDescriptor::Unknown;
142 }
143#endif // defined(_WIN32)
144
145 return Err(SystemError::fromErrno());
146}

178inline auto truncate(fd_t fd, uint64_t size) -> IoResult<void> {
179
180#if defined(_WIN32)

182 ::LARGE_INTEGER newSize { .QuadPart = static_cast<::LONGLONG>(size) };
183 auto ok = ::SetFilePointerEx(fd, newSize, &currentPos, FILE_BEGIN) && ::SetEndOfFile(fd);
184 ::SetFilePointerEx(fd, currentPos, nullptr, FILE_BEGIN);
185 if (ok) {
186 return {};
187 }
188#else
189 if (::ftruncate(fd, size) == 0) {
