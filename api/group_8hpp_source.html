135 using Vector = std::vector<Value>;
136
137 auto await_resume() -> Vector {
138 Vector vec;
139 while (mGroup.completionSize() > 0) { // Collect all completions
140 auto ctxt = mGroup.nextCompletion();
141 auto val = static_cast<TaskSpawnContext<T> &>(*ctxt).value();
142 if (val) { // Is not stopped
143 vec.emplace_back(std::move(*val));
