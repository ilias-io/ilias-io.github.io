133 constexpr auto gray = "\033[90m";
134 constexpr auto green = "\033[1;32m"; // Bold Green for Function
135 constexpr auto cyan = "\033[36m"; // Cyan for File

139 auto str = std::string {};
140 auto idx = size_t {0};
141 auto append = [&](std::string_view content, std::string_view color = {}) {
142 if (color.empty()) {
143 str += content;
144 return;
145 }
146 if (mode == Color) {
147 str += color;
148 str += content;
149 str += rst;
150 }
151 else {
152 str += content;
153 }
154 };
155 str.reserve(mFrames.size() * 64);
156 if (mFrames.empty()) {
157 str = "<Empty Stacktrace>";
158 return str;
159 }
160
161 for (auto &frame : mFrames) {
162 append("#", gray);
163 append(std::to_string(idx), gray);
164 append(" ");
165 append(frame.function(), green);
166 append("\n");
167
168 append(" at ", gray);
169 append(frame.filename(), cyan);
170 append(":", gray);
171 append(std::to_string(frame.line()), yellow);
172
173 auto msg = frame.message();
174 if (!msg.empty()) {
175 append(" [", gray);
176 append(msg, magenta);
177 append("]", gray);
178 }
179
180 str += "\n";
181 idx += 1;
182 }
183 return str;
184 }
