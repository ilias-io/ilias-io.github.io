92template <typename T>
93class RecvAwaiter {
94public:
95 RecvAwaiter(ChanReceiver<T> chan) : mChan(std::move(chan)), mLocker(*mChan) { } // GOT ownship and lock here
96 RecvAwaiter(RecvAwaiter &&other) = default;
97 ~RecvAwaiter() = default;
98
99 auto await_ready() const -> bool {
100 // Already sended or broken
101 return mChan->value.has_value() || mChan->senderClose;
102 }
103
104 auto await_suspend(runtime::CoroHandle caller) -> void {
105 mChan->receiver = caller;
106 mLocker.unlock(); // Unlock the channel, lock again when resume or stopRequested
107 mReg.register_<&RecvAwaiter::onStopRequested>(caller.stopToken(), this);
108 }
109
110 auto await_resume() -> std::optional<T> {
111 if (!mLocker.owns_lock()) {
112 mLocker.lock();
113 }
114 ILIAS_ASSERT(!mChan->valueGot, "Double call on recv ?, value already got");
115 ILIAS_ASSERT(!mChan->receiver);
116
117 if (mChan->value) { // Value is set
118 mChan->valueGot = true;
119 return std::move(*mChan->value);
120 }
121 ILIAS_ASSERT(mChan->senderClose); // Should be closed, if not, wrong state
122 return std::nullopt;
123 }
124private:
125 auto onStopRequested() -> void { // On the executor thread
126 mLocker.lock();
127 auto handle = std::exchange(mChan->receiver, nullptr);
128 mLocker.unlock();
129
130 if (handle) { // Not already scheduled
131 handle.setStopped();
132 }
133 }
134
135 ChanReceiver<T> mChan;
136 std::unique_lock<Channel<T> > mLocker; // Lock the channel, unlock before destroy the chan
137 runtime::StopRegistration mReg;
138};
