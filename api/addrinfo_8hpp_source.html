173inline auto AddressInfo::lookup(std::string_view endpoint, std::optional<addrinfo_t> hints) -> IoTask<AddressInfo> {
174 // Split xxx:port
175 std::string_view name;
176 std::string_view service;
177 if (auto sep = endpoint.rfind(':'); sep != std::string_view::npos) {
178 name = endpoint.substr(0, sep);
179 service = endpoint.substr(sep + 1);
180
181 // Handle V6
182 // [::]:666
183 if (name.starts_with('[') && name.ends_with(']')) {
184 name.remove_prefix(1);
185 name.remove_suffix(1);
186 }
187 }
188 else {
189 name = endpoint;
190 }
191 return AddressInfo::fromHostname(name, service, hints);
192}
