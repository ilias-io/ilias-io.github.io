85template <int Level, int Type, std::default_initializable T> requires(std::is_trivially_copyable_v<T>)
86class CmsgT {
87public:
88 constexpr CmsgT() = default;
89 constexpr CmsgT(T value) : mValue{value} {}
90
91 // Properties
92 static auto level() noexcept -> int { return Level; }
93 static auto type() noexcept -> int { return Type; }
94
95 // Deserialization
96 static auto decode(Buffer buffer) -> std::optional<CmsgT> {
97 if (buffer.size() != sizeof(T)) {
98 return std::nullopt;
99 }
100 T val{};
101 std::memcpy(&val, buffer.data(), sizeof(T));
102 return CmsgT{val};
103 }
104
105 // Serialization
106 auto encode() const -> Buffer {
107 return makeBuffer(&mValue, sizeof(T));
108 }
109
110 // Get the inner value
111 constexpr auto value() const noexcept -> T { return mValue; }
112 constexpr operator T() const noexcept { return mValue; }
113private:
114 T mValue {};
115};

227 auto resize(size_t size) -> void {
228 // In windows, check the size is not too large
229#if defined(_WIN32)
230 ILIAS_ASSERT(std::in_range<decltype(::WSABUF::len)>(size), "The size {} is too large, can't passed by ABI", size);
231#endif // _WIN32
232
233 if (size == 0) {
234 mBuffer.reset();
235 mSize = 0;
236 return;
237 }
238 auto *ptr = static_cast<std::byte*>(std::realloc(mBuffer.get(), size));
239 if (!ptr) [[unlikely]] {
240 ILIAS_THROW(std::bad_alloc{});
241 }
242 mBuffer.release();
243 mBuffer.reset(ptr);
244 mSize = size;
245 }
246
252 auto data() -> MutableBuffer {
253 return {mBuffer.get(), mSize};
254 }

263 }

270 std::free(ptr);
271 }
272 };

303 template <cmsg::EncodeCmsg T>
304 auto append(const T &msg) -> CmsgWriter & {
305 auto encoded = msg.encode();
306 auto data = makeBuffer(encoded); // The encoded may a vector, cast it to view
307 auto spaceSize = ILIAS_CMSG_SPACE(data.size());
308
309 // Check if there is enough space
310 if (mSpace.size() < mHead + spaceSize) [[unlikely]] {
311 mSpace.resize(mHead + spaceSize);
312 }
313
314 // Get the storage space and zero it
315 auto *storage = mSpace.data().data() + mHead;
316 std::memset(storage, 0, spaceSize);
317
318 // Construct the header
319 auto *hdr = new (storage) cmsghdr_t {
320 .cmsg_len = ILIAS_CMSG_LEN(data.size()),
321 .cmsg_level = T::level(),
322 .cmsg_type = T::type(),
323 };
324
325 // Copy the data
326 std::memcpy(ILIAS_CMSG_DATA(hdr), data.data(), data.size());
327
328 // Advance the head
329 mHead += spaceSize;
330 return *this;
331 }
332
338 auto end() -> Buffer {
339 if (mHead == 0) {
340 // Winsock requires a null control pointer when no control messages are sent.
341 return {};
342 }
343 return mSpace.data().subspan(0, mHead);
344 }
345private:
346 CmsgSpace &mSpace;
347 size_t mHead = 0;
348};
