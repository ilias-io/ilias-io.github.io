261 // Seekable
269 auto seek(int64_t offset, SeekOrigin origin) -> IoTask<uint64_t> {
270 if (!mOffset) {
271 co_return Err(IoError::OperationNotSupported);
272 }
273 int64_t now = *mOffset;
274 switch (origin) {
275 case SeekOrigin::Begin: now = offset; break;
276 case SeekOrigin::Current: now += offset; break;
277 case SeekOrigin::End: {
278 ILIAS_CO_TRY(auto s, co_await size());
279 now = s + offset;
280 }
281 }
282 mOffset = std::max<int64_t>(0, now);
283 co_return *mOffset;
284 }

359inline auto OpenOptions::doOpen(OpenOptions self, fs::Path path) -> IoTask<File> {
360 // Validate
361 if (!self.mRead && !self.mWrite && !self.mAppend) {
362 co_return Err(IoError::InvalidArgument);
363 }
364
365 // Truncate need write permission
366 if (self.mTruncate && !self.mWrite) {
367 co_return Err(IoError::InvalidArgument);
368 }
369
370 // Append is conflict with truncate
371 if (self.mAppend && self.mTruncate) {
372 co_return Err(IoError::InvalidArgument);
373 }
374
375 // Create need write permission
376 if ((self.mCreate || self.mCreateNew) && !self.mWrite && !self.mAppend) {
377 co_return Err(IoError::InvalidArgument);
378 }
379#if defined(_WIN32)
380 // create | truncate | createNew | disposition
381 // --------|----------|------------|------------------
382 // - | - | - | OPEN_EXISTING
383 // ✓ | - | - | OPEN_ALWAYS
384 // - | ✓ | - | TRUNCATE_EXISTING
385 // ✓ | ✓ | - | CREATE_ALWAYS
386 // * | * | ✓ | CREATE_NEW
387 ::DWORD access = 0;
388 ::DWORD creationDisposition = 0;
389 ::DWORD flagsAndAttributes = FILE_FLAG_OVERLAPPED;
390 ::DWORD lastError = ERROR_SUCCESS;
391
392 // Access
393 if (self.mRead) {
394 access |= GENERIC_READ;
395 }
396 if (self.mWrite) {
397 access |= GENERIC_WRITE;
398 }
399 if (self.mAppend) {
400 access &= ~static_cast<::DWORD>(GENERIC_WRITE);
401 access |= FILE_APPEND_DATA;
402 }
403
404 // creationDisposition
405 if (self.mCreateNew) {
406 creationDisposition = CREATE_NEW;
407 }
408 else if (self.mCreate && self.mTruncate) {
409 creationDisposition = CREATE_ALWAYS;
410 }
411 else if (self.mCreate) {
412 creationDisposition = OPEN_ALWAYS;
413 }
414 else if (self.mTruncate) {
415 creationDisposition = TRUNCATE_EXISTING;
416 }
417 else {
418 creationDisposition = OPEN_EXISTING;
419 }
420
421 ::HANDLE fd = co_await ilias::blocking([&]() { // The CreateFileW may block
422 ::HANDLE fd = ::CreateFileW(
423 path.wstring().c_str(),
424 access,
425 self.mShareMode,
426 nullptr,
427 creationDisposition,
428 flagsAndAttributes,
429 nullptr
430 );
431 if (fd == INVALID_HANDLE_VALUE) {
432 lastError = ::GetLastError();
433 }
434 return fd;
435 });
436 if (lastError != ERROR_SUCCESS) {
437 co_return Err(SystemError(lastError));
438 }
439#else
440 // Mapping by man fopen
441 // r | O_RDONLY
442 // w | O_WRONLY | O_CREAT | O_TRUNC
443 // a | O_WRONLY | O_CREAT | O_APPEND
444 // r+ | O_RDWR
445 // w+ | O_RDWR | O_CREAT | O_TRUNC
446 // a+ | O_RDWR | O_CREAT | O_APPEND
447 int flags = O_CLOEXEC;
448 int errc = 0;
449 bool write = self.mWrite || self.mAppend;
450
451 if (self.mRead && write) {
452 flags |= O_RDWR;
453 }
454 else if (self.mRead) {
455 flags |= O_RDONLY;
456 }
457 else if (write) {
458 flags |= O_WRONLY;
459 }
460
461 if (self.mAppend) {
462 flags |= O_APPEND;
463 }
464 if (self.mTruncate) {
465 flags |= O_TRUNC;
466 }
467 if (self.mCreate) {
468 flags |= O_CREAT;
469 }
470 if (self.mCreateNew) {
471 flags |= O_CREAT | O_EXCL;
472 }
473
474 auto fd = co_await ilias::blocking([&]() {
475 auto u8 = path.u8string();
476 auto s = reinterpret_cast<const char *>(u8.c_str());
477 auto fd = 0;
478 if (flags & O_CREAT) {
479 fd = ::open(s, flags, self.mMode);
480 }
481 else {
482 fd = ::open(s, flags);
483 }
484 if (fd < 0) {
485 errc = errno;
486 }
487 return fd;
488 });
489 if (fd < 0) {
490 co_return Err(SystemError(errc));
491 }
492#endif // defined(_WIN32)
493
494 // Wrap the file descriptor
495 ILIAS_CO_TRY(auto handle, IoHandle<FileDescriptor>::make(FileDescriptor {fd}, IoDescriptor::File));
496
497 std::optional<uint64_t> off;
498 // Check if the file is a regular file (support offset)
499 // Copy the lowlevel offset from the fd, and let us manage it
500 if (self.mAppend) { // Append mode, the offset is not available
501 co_return File {std::move(handle), std::nullopt};
502 }
503#if defined(_WIN32)
504 if (::GetFileType(fd) == FILE_TYPE_DISK) {
505 ::LARGE_INTEGER offset { .QuadPart = 0 };
506 ::LARGE_INTEGER cur;
507 if (::SetFilePointerEx(fd, offset, &cur, FILE_CURRENT)) { // NOLINT(readability-implicit-bool-conversion)
508 off = cur.QuadPart;
509 }
510 }
511#else
512 struct stat st;
513 if (::fstat(fd, &st) == 0 && S_ISREG(st.st_mode)) {
514 auto offset = ::lseek(fd, 0, SEEK_CUR);
515 if (offset != -1) {
516 off = offset;
517 }
518 }
519#endif // defined(_WIN32)
520
521 co_return File{std::move(handle), off};
522}
