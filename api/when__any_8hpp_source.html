104template <typename ...Ts>
105class WhenAnyAwaiter final : public WhenAnyAwaiterBase {
106public:
107 using WhenAnyAwaiterBase::WhenAnyAwaiterBase; // Inherit the constructor
108
109 using Tuple = std::tuple<Option<Ts> ...>; // Using Option to replace void to std::monostate
110 using RawTuple = std::tuple<Ts...>;
111
112 auto await_resume() -> Tuple {
113 ILIAS_ASSERT(mGot, "No value produced, but await_resume called?, ??? INTERNAL BUG");
114 return makeResult(std::make_index_sequence<sizeof...(Ts)>{});
115 }
116private:
117 template <size_t I>
118 auto makeResult() -> std::tuple_element_t<I, Tuple> {
119 using Raw = std::tuple_element_t<I, RawTuple>;
120 if (mGot == &mTasks[I]) {
121 return makeOption([&]() { // We replace void to std::monostate in Tuple, but in here, we should use TaskHandle<void>, not std::monostate
122 auto handle = TaskHandle<Raw>::cast(mTasks[I].task());
123 return handle.value();
124 });
125 }
126 return std::nullopt;
127 }
128
129 template <size_t ...Is>
130 auto makeResult(std::index_sequence<Is...>) -> Tuple {
131 return std::tuple {makeResult<Is>()...};
132 }
133};

175inline auto whenAny([[ILIAS_CORO_ELIDABLE_ARGUMENT]] Ts && ...args) {
176 return task::WhenAnyTuple<AwaitableResult<Ts>... > { // Construct the task for the given awaitable

189template <AwaitableSequence T>
190[[nodiscard]]
191inline auto whenAny([[ILIAS_CORO_ELIDABLE_ARGUMENT]] T seq) {
192 std::vector<task::TaskContext> tasks{};
193 if constexpr (std::ranges::sized_range<T>) {
194 tasks.reserve(std::ranges::size(seq));
195 }
196 for (auto &awaitable : seq) { // Construct the task for the given awaitable to an vector
197 tasks.emplace_back(toTask(std::move(awaitable))._leak());
198 }
199 if (tasks.empty()) { // Contract violation
200 ILIAS_THROW(std::invalid_argument{"Can't whenAny on empty sequence"});
201 }
202 return task::WhenAnySequence<AwaitableSequenceValue<T> > {
203 .mTasks = std::move(tasks),
204 .mContext = nullptr,
205 };
206}
