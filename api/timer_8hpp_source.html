140inline auto TimerService::updateTimers() -> void {
141 if (mTimers.empty()) {
142 return;
143 }
144 auto now = std::chrono::steady_clock::now();
145 auto changed = false;
146 for (auto iter = mTimers.begin(); iter != mTimers.end(); ) {
147 auto &[timepoint, awaiter] = *iter;
148 if (timepoint > now) {
149 break;
150 }
151 ILIAS_TRACE("TimerSevice", "Submit timer at {}, diff {}, awaiter {}", timepoint.time_since_epoch(), now - timepoint, (void*) awaiter);
152 awaiter->onTimeout();
153 changed = true;
154 iter = mTimers.erase(iter);
155 }
156 if (changed) {
157 timerChanged();
158 }
159}

182inline auto TimerService::timerChanged() -> void {
