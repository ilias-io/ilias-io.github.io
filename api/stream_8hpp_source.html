79 auto prepare(size_t size) -> MutableBuffer {
80 if (mPos == mTail) { //< Input window is empty, move it and us to the begin of the buffer
81 mPos = 0;
82 mTail = 0;
83 }
84 if ((mTail - mPos) < mBuffer.size() / 8) { //< If the input window is too small, move it to the begin of the buffer
85 ::memmove(mBuffer.data(), mBuffer.data() + mPos, mTail - mPos);
86 mTail -= mPos;
87 mPos = 0;
88 }
89 if ((mTail - mPos) + size > mMaxCapacity) { //< Reach the limit
90 return {};
91 }
92 auto space = mBuffer.size() - mTail; // The space left in the buffer
93 if (space < size) { //< Reallocate the buffer if there is not enough space
94 auto newSize = std::min((mBuffer.size() + size) * 2, mMaxCapacity);
95 auto newSpace = newSize - mTail;
96 if (newSpace < size) { // The new buffer is still not enough, we reach the limit
97 return {};
98 }
99 mBuffer.resize(newSize);
100 }
101 return std::span{mBuffer}.subspan(mTail, size);
102 }
103
109 auto commit(size_t size) -> void {
110 ILIAS_ASSERT(size <= (mBuffer.size() - mTail), "Commit size {} exceed the capacity {}", size, mBuffer.size() - mTail);
111 size = std::min(size, mBuffer.size() - mTail); //< In release version, the assert may be removed, so we add it
112 mTail += size;
113 }

376public:
383 BufReader(T stream, size_t capacity = DEFAULT_BUFFER_CAPACITY) : mBuffer(capacity), mStream(std::move(stream)) {}
384 BufReader(BufReader &&) = default;
385 BufReader() = default;
386
387 // Readable
388 auto read(MutableBuffer buffer) -> IoTask<size_t> {
389 // No data cached && This buffer is much bigger, read directly
390 if (mBuffer.empty() && buffer.size() >= mBuffer.maxCapacity() / 2) {
391 co_return co_await mStream.read(buffer);
392 }
393 auto data = co_await fill();
394 if (!data) {
395 co_return Err(data.error());
396 }
397 if (data->empty()) { // EOF
398 co_return 0;
399 }
400 auto size = std::min(buffer.size(), data->size());
401 ::memcpy(buffer.data(), data->data(), size);
402 mBuffer.consume(size);
403 co_return size;
404 }
405
406 // BufReadable
414 auto fill(FillPolicy policy = FillPolicy::None) -> IoTask<Buffer> {
415 if (mBuffer.empty() || policy == FillPolicy::More) {
416 auto bufsize = mBuffer.maxCapacity() - mBuffer.size(); // Get the remaining capacity of the buffer
417 if (bufsize == 0 && policy == FillPolicy::More) {
418 co_return Err(IoError::NoBufferSpaceAvailable); // Failed to fill the buffer with more data
419 }
420 auto buf = mBuffer.prepare(bufsize);
421 auto res = co_await mStream.read(buf);
422 if (!res) {
423 co_return Err(res.error());
424 }
425 if (res == 0 && policy == FillPolicy::More) {
426 co_return Err(IoError::UnexpectedEOF); // Failed to fill the buffer with more data
427 }
428 mBuffer.commit(*res);
429 }
430 co_return mBuffer.data();
431 }
432
433 // Consume the data of the buffer
434 auto consume(size_t size) -> void {
435 return mBuffer.consume(size);
436 }

441 }

490class BufWriter final : public StreamExt<BufWriter<T> > {
491public:
498 BufWriter(T stream, size_t capacity = DEFAULT_BUFFER_CAPACITY) : mBuffer(capacity), mStream(std::move(stream)) {}
499 BufWriter(BufWriter &&) = default;
500 BufWriter() = default;
501
502 // Writable
503 auto write(Buffer buffer) -> IoTask<size_t> {
504 // Bigger than capacity / 2, write directly
505 if (buffer.size() >= mBuffer.maxCapacity() / 2) {
506 ILIAS_CO_TRYV(co_await flush());
507 co_return co_await mStream.write(buffer);
508 }
509 // No room in buffer
510 if (mBuffer.maxCapacity() - mBuffer.size() < buffer.size()) {
511 ILIAS_CO_TRYV(co_await flush());
512 }
513 // Copy into buffer
514 auto data = mBuffer.prepare(buffer.size());
515 ::memcpy(data.data(), buffer.data(), buffer.size());
516 mBuffer.commit(buffer.size());
517 co_return buffer.size();
518 }
519
520 auto flush() -> IoTask<void> {
521 while (!mBuffer.empty()) {
522 ILIAS_CO_TRY(auto n, co_await mStream.write(mBuffer.data()));
523 if (n == 0) {
524 co_return Err(IoError::WriteZero);
525 }
526 mBuffer.consume(n);
527 }
528 co_return co_await mStream.flush();
529 }
530
531 auto shutdown() -> IoTask<void> {
532 ILIAS_CO_TRYV(co_await flush());
533 co_return co_await mStream.shutdown();
534 }

538 return mStream.read(buffer);
539 }
