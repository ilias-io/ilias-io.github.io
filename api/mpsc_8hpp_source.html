40template <Sendable T>
41class Channel final : public sync::ChannelBase {
42public:
43 Channel(size_t c) : capacity(c) { }
44 Channel(Channel &&) = delete;
45 ~Channel() { // Check state
46 ILIAS_ASSERT(receiverClosed);
47 ILIAS_ASSERT(senderClosed);
48 }
49
50 // For sender
51 auto trySendInternal(Result<void, T> &item) -> bool {
52 std::scoped_lock locker{mutex};
53 if (receiverClosed) { // We can't send any more data.
54 return true;
55 }
56 if (queue.size() + reserved >= capacity) {
57 return false; // No space, continue to wait.
58 }
59 queue.emplace_back(std::move(item.error()));
60 item = {};
61 return true; // Sended
62 }
63
64 // Returns true when wait is finished. outGot is true only when a slot was reserved.
65 auto tryReserveInternal(bool &outGot) -> bool {
66 std::scoped_lock locker{mutex};
67 if (receiverClosed) { // We can't send any more data.
68 outGot = false;
69 return true;
70 }
71 if (queue.size() + reserved >= capacity) {
72 return false; // No space, continue to wait.
73 }
74 ++reserved;
75 outGot = true;
76 return true; // Reserved
77 }
78
79 auto onSenderClose() -> void {
80 bool notify = false;
81 {
82 std::scoped_lock locker{mutex};
83 senderClosed = true; // All sender is closed.
84 notify = !receiverClosed; // If the receiver is closed, not need to notify.
85 }
86 if (notify) { // Receiver is not closed, we should wakeup the receiver.
87 receiver.wakeupOne();
88 }
89 }
90
91 // For receiver
92 auto tryRecvInternal(std::optional<T> &value) -> bool {
93 std::scoped_lock locker{mutex};
94 if (!queue.empty()) {
95 value.emplace(std::move(queue.front()));
96 queue.pop_front();
97 return true; // Have data
98 }
99 if (senderClosed) {
100 return true; // No sender, so we can't receive any more.
101 }
102 return false;
103 }
104
105 auto onReceiverClose() -> void {
106 bool notify = false;
107 {
108 std::scoped_lock locker{mutex};
109 receiverClosed = true; // All receiver is closed.
110 notify = !senderClosed; // If the sender is all closed, not need to notify.
111 }
112 if (notify) {
113 senders.wakeupAll(); // Because it is multi producer, use wakeupAll
114 }
115 }
116
117
118 // States
119 const size_t capacity {0}; // The capacity of the channel. read only.
120 bool senderClosed {false}; // If all the sender is closed.
121 bool receiverClosed {false};
122
123 // Sync, all queue's wakeup must call without lock the mutex, we may lock the mutex in the onWakeup. it will deadlock.
124 sync::FutexMutex mutex; // Protect the queue, reserved, senderCount, and receiverClosed.
125 sync::WaitQueue senders;
126 sync::WaitQueue receiver;
127 std::deque<T> queue; // For storing the items.
128 size_t reserved {0}; // The num of reserved items
129};

371 std::unique_lock locker{mChan->mutex};
372
373 // Do check
374 if (mChan->receiverClosed) {
375 return Err(TrySendErrorResult<T>{ .item = std::move(item), .reason = TrySendError::Closed });

378 return Err(TrySendErrorResult<T>{ .item = std::move(item), .reason = TrySendError::Full });
379 }
380 mChan->queue.emplace_back(std::move(item));
381 locker.unlock();

383 // Success to send, wakeup the receiver.

538 std::unique_lock locker{mChan->mutex};
539 if (mChan->queue.empty() && mChan->senderClosed) {
540 return Err(TryRecvError::Closed);
541 }
542 if (mChan->queue.empty()) {
543 return Err(TryRecvError::Empty);
544 }

546 mChan->queue.pop_front();
547 locker.unlock();
548
549 // Success to recv, wakeup the sender.
550 mChan->senders.wakeupOne();
551 return std::move(value);
552 }
