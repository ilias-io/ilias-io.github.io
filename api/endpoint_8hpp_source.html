461 static auto fromString(std::string_view buffer) -> Result<IPEndpoint, std::errc> {
462 // Split to addr and port
463 auto pos = buffer.rfind(':');
464 if (pos == std::string_view::npos || pos < 4) { // 4 is the minimum length of [::]:port
465 return Err(std::errc::invalid_argument);
466 }
467
468 // Parse the port
469 auto portStr = buffer.substr(pos + 1);
470 uint16_t port = 0;
471 if (auto [ptr, ec] = std::from_chars(portStr.data(), portStr.data() + portStr.size(), port); ec != std::errc{}) {
472 return Err(ec);
473 }
474 else if (ptr != portStr.data() + portStr.size()) { // The port may contain non-numeric characters like [::]:20char
475 return Err(std::errc::invalid_argument);
476 }
477
478 // Parse the address
479 auto addrStr = buffer.substr(0, pos);
480 if (addrStr.front() == '[' && addrStr[pos - 1] == ']') {
481 // IPV6
482 addrStr = addrStr.substr(1, pos - 2);
483 }
484 ILIAS_TRY(auto addr, IPAddress::fromString(addrStr));
485 return IPEndpoint{addr, port};
486 }
