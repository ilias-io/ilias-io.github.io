42template <Sendable T>
43class Channel final : public sync::ChannelBase {
44public:
45 Channel(size_t c) : capacity(c) {}
46 Channel(Channel &&) = delete;
47 ~Channel() { // Check state
48 ILIAS_ASSERT(receiverClosed);
49 ILIAS_ASSERT(senderClosed);
50 }
51
52 // For sender
53 auto trySendInternal(Result<void, T> &item) -> bool {
54 std::scoped_lock locker{mutex};
55 if (receiverClosed) { // We can't send any more data.
56 return true;
57 }
58 if (queue.size() + reserved >= capacity) {
59 return false; // No space, continue to wait.
60 }
61 queue.emplace_back(std::move(item.error()));
62 item = {};
63 return true; // Sended
64 }
65
66 // Returns true when wait is finished. outGot is true only when a slot was reserved.
67 auto tryReserveInternal(bool &outGot) -> bool {
68 std::scoped_lock locker{mutex};
69 if (receiverClosed) { // We can't send any more data.
70 outGot = false;
71 return true;
72 }
73 if (queue.size() + reserved >= capacity) {
74 return false; // No space, continue to wait.
75 }
76 ++reserved;
77 outGot = true;
78 return true; // Reserved
79 }
80
81 auto onSenderClose() -> void {
82 bool notify = false;
83 {
84 std::scoped_lock locker{mutex};
85 senderClosed = true; // All sender is closed.
86 notify = !receiverClosed; // If the receiver is closed, not need to notify.
87 }
88 if (notify) { // Wake all receivers so they can observe the closed state.
89 receivers.wakeupAll();
90 }
91 }
92
93 // For receiver
94 auto tryRecvInternal(std::optional<T> &value) -> bool {
95 std::scoped_lock locker{mutex};
96 if (!queue.empty()) {
97 value.emplace(std::move(queue.front()));
98 queue.pop_front();
99 return true; // Have data
100 }
101 if (senderClosed) {
102 return true; // No sender, so we can't receive any more.
103 }
104 return false;
105 }
106
107 auto onReceiverClose() -> void {
108 bool notify = false;
109 {
110 std::scoped_lock locker{mutex};
111 receiverClosed = true; // All receiver is closed.
112 notify = !senderClosed; // If the sender is all closed, not need to notify.
113 }
114 if (notify) {
115 senders.wakeupAll(); // Multi producer, use wakeupAll
116 }
117 }
118
119 // States
120 const size_t capacity {0}; // The capacity of the channel. read only.
121 bool senderClosed {false}; // If all the sender is closed.
122 bool receiverClosed {false}; // If all the receiver is closed.
123
124 // Sync, all queue's wakeup must call without lock the mutex, we may lock the mutex in the onWakeup. it will deadlock.
125 sync::FutexMutex mutex; // Protect the queue, reserved, and closed flags.
126 sync::WaitQueue senders;
127 sync::WaitQueue receivers;
128 std::deque<T> queue; // For storing the items.
129 size_t reserved {0}; // The num of reserved items
130};
