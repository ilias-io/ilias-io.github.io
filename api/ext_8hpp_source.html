80template <Writable T>
81inline auto writeAll(T &stream, Buffer buffer) -> IoTask<size_t> {
82 size_t written = 0;
83 while (!buffer.empty()) {
84 ILIAS_CO_TRY(auto n, co_await stream.write(buffer));
85 if (n == 0) {
86 co_return Err(IoError::WriteZero);
87 }
88 written += n;
89 buffer = buffer.subspan(n);
90 }
91 co_return written;
92}

164template <Readable T, MemWritable Container>
165inline auto readToEnd(T &stream, Container &container) -> IoTask<size_t> {
166 size_t size = container.size();
167 size_t readed = 0;
168 while (true) {
169 container.resize(size + 1024);
170 auto buf = makeBuffer(container).subspan(size);
171 while (!buf.empty()) {
172 auto n = co_await stream.read(buf);
173 if (!n) { // Error resize the back
174 container.resize(size);
175 co_return Err(n.error());
176 }
177 if (*n == 0) { // EOF
178 container.resize(size);
179 co_return readed;
180 }
181 readed += *n;
182 size += *n;
183 buf = buf.subspan(*n);
184 }
185 }
186}

203// MARK: BufReadable
213template <BufReadable T>
214inline auto readline(T &stream, std::string &str, std::string_view delim = "\n") -> IoTask<size_t> {
215 size_t readed = 0;
216 while (true) {
217 ILIAS_CO_TRY(auto buf, co_await stream.fill(FillPolicy::None));
218 if (buf.empty() && readed != 0) { // EOF, but we have readed some data
219 co_return readed;
220 }
221 if (buf.empty()) { // EOF
222 co_return Err(IoError::UnexpectedEOF);
223 }
224 auto view = std::string_view {reinterpret_cast<const char *>(buf.data()), buf.size()};
225 auto pos = view.find(delim);
226 if (pos == std::string_view::npos) {
227 str.append(view);
228 stream.consume(view.size());
229 readed += view.size();
230 continue;
231 }
232 else {
233 auto sub = view.substr(0, pos + delim.size());
234 str.append(sub);
235 stream.consume(sub.size());
236 readed += sub.size();
237 break;
238 }
239 }
240 co_return readed;
241}

272template <Writable T, Readable U>
273inline auto copy(T &dst, U &src) -> IoTask<size_t> {
274 std::byte buffer[1024 * 8]; // 8KB buffer, change the size if needed
275 size_t written = 0;
276 while (true) {
277 ILIAS_CO_TRY(auto readed, co_await src.read(buffer));
278 if (readed == 0) { // EOF
279 break;
280 }
281 ILIAS_CO_TRYV(co_await io::writeAll(dst, std::span{buffer}.subspan(0, readed)));
282 written += readed;
283 }
284 co_return written;
285}

294template <Layer T>
295inline auto lowestLayer(T &layer) -> decltype(auto) {
296 auto walk = [](auto self, auto &cur) -> decltype(auto) {
297 if constexpr (Layer<decltype(cur)>) {
298 return self(self, cur.nextLayer());
299 }
300 else {
301 return cur;
302 }
303 };
304 return walk(walk, layer);
