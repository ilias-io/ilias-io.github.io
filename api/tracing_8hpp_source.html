246// MARK: TracingAwaitable
247// Add hooks to an awaitable, used for tracing
248template <typename T, bool Forward>
249class TracingAwaitable {
250public:
251 template <typename U>
252 using DecayIf = std::conditional_t<Forward, U, std::decay_t<U> >; // If not forward, move the awaitable
253 using Awaitable = DecayIf<T>;
254 using Awaiter = DecayIf<decltype(toAwaiter(std::declval<T>()))>;
255
256 // Store the awaitable and make the awaiter by forward (move on !Forward), source is the location of the await point
257 TracingAwaitable(T awaitable, TraceContext &ctxt, CaptureSource source) :
258 mAwaitable(std::forward<T>(awaitable)),
259 mAwaiter(toAwaiter(std::forward<T>(mAwaitable))),
260 mCtxt(ctxt),
261 mSource(source) {}
262
263 // MUST RVO, Pin the awaitable, avoid the awaiter implementation need an stable awaitable address, it will cause dangling if move
264 TracingAwaitable(const TracingAwaitable &) = delete;
265
266 // Hooks
267 auto await_ready() noexcept(noexcept(mAwaiter.await_ready())) -> bool {
268 // Update the line number of the top frame
269 // co_await doStd();
270 if (auto *frame = mCtxt.topFrame(); frame) {
271 frame->setLine(mSource.toLocation().line());
272 }
273 return mAwaiter.await_ready();
274 }
275
276 template <typename U>
277 auto await_suspend(std::coroutine_handle<U> handle) noexcept(noexcept(mAwaiter.await_suspend(handle))) {
278 using Ret = decltype(mAwaiter.await_suspend(handle));
279 if constexpr (std::is_same_v<Ret, void>) {
280 mAwaiter.await_suspend(handle);
281 mCtxt.suspend(mSource);
282 return;
283 }
284 else if constexpr (std::convertible_to<Ret, bool>) { // Return bool
285 auto ret = mAwaiter.await_suspend(handle);
286 if (ret) { // true on actually suspend
287 mCtxt.suspend(mSource);
288 }
289 return ret;
290 }
291 else { // std::coroutine_handle<>
292 auto ret = mAwaiter.await_suspend(handle);
293 mCtxt.suspend(mSource);
294 return ret;
295 }
296 }
297
298 auto await_resume() noexcept(noexcept(mAwaiter.await_resume())) -> decltype(auto) {
299 mCtxt.resume(mSource);
300 return mAwaiter.await_resume();
301 }
302private:
303 Awaitable mAwaitable;
304 Awaiter mAwaiter;
305 TraceContext &mCtxt;
306 CaptureSource mSource;
307};
