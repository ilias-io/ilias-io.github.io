218 template <typename Fn>
219 static auto withOpen(std::string_view file, Fn fn) -> bool {
220 ::FILE *fp = nullptr;
221#if defined(_WIN32)
222 // Use wopen for unicode, and use _s version to let the compiler shut up
223 auto err = ::_wfopen_s(&fp, win32::toWide(file).c_str(), L"rb");
224 if (err != 0) {
225 return false;
226 }
227#else
228 fp = ::fopen(std::string(file).c_str(), "rb");
229#endif // defined(_WIN32)
230 if (!fp) {
231 return false;
232 }
233 ScopeExit _([fp]() {
234 ::fclose(fp);
235 });
236
237 ::fseek(fp, 0, SEEK_END);
238 auto size1 = ::ftell(fp);
239 if (size1 <=0) {
240 return false;
241 }
242 ::fseek(fp, 0, SEEK_SET);
243
244 auto size = static_cast<size_t>(size1);
245 auto ptr = std::make_unique<std::byte []>(size);
246 auto n = ::fread(ptr.get(), 1, size, fp);
247 if (n != size) {
248 return false;
249 }
250 return fn(std::span{ptr.get(), n});
251 }
