443inline auto TcpBuilder::bind(IPEndpoint endpoint, int backlog) -> Just<IoResult<TcpListener> > {
444 return just([&]() -> IoResult<TcpListener> {
445 ILIAS_TRY(auto sockfd, std::move(mFd));
446 ILIAS_TRYV(sockfd.bind(endpoint));
447 ILIAS_TRYV(sockfd.listen(backlog));
448 ILIAS_TRY(auto handle, IoHandle<Socket>::make(std::move(sockfd), IoDescriptor::Socket));
449 return TcpListener {std::move(handle)};
450 }());
451}

453inline auto TcpBuilder::impl(TcpBuilder self, IPEndpoint endpoint) -> IoTask<TcpStream> {
454 ILIAS_CO_TRY(auto sockfd, std::move(self.mFd));
455 ILIAS_CO_TRY(auto handle, IoHandle<Socket>::make(std::move(sockfd), IoDescriptor::Socket));
456 ILIAS_CO_TRYV(co_await handle.connect(endpoint));
457 co_return TcpStream {std::move(handle)};
